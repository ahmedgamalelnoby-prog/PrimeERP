from sqlalchemy.orm import relationship
from sqlalchemy.sql import func

from app.database.base import Base


role_permissions = Table(
    "role_permissions",
    Base.metadata,
    Column(
        "role_id",
        Integer,
        ForeignKey(
            "roles.id",
            ondelete="CASCADE",
        ),
        primary_key=True,
    ),
    Column(
        "permission_id",
        Integer,
        ForeignKey(
            "permissions.id",
            ondelete="CASCADE",
        ),
        primary_key=True,
    ),
)


class Role(Base):
    __tablename__ = "roles"

    id = Column(
        Integer,
        primary_key=True,
        index=True,
    )

    name = Column(
        String(50),
        unique=True,
        nullable=False,
        index=True,
    )

    description = Column(
        String(255),
        nullable=True,
    )

    is_active = Column(
        Boolean,
        default=True,
        nullable=False,
    )

    created_at = Column(
        TIMESTAMP,
        server_default=func.now(),
        nullable=False,
    )

    updated_at = Column(
        TIMESTAMP,
        server_default=func.now(),
        onupdate=func.now(),
        nullable=False,
    )

    users = relationship(
        "User",
        back_populates="role",
        foreign_keys="User.role_id",
    )

    permissions = relationship(
        "Permission",
        secondary=role_permissions,
        back_populates="roles",
    )

    def __repr__(self) -> str:
        return f"<Role {self.name}>"


class Permission(Base):
    __tablename__ = "permissions"

    action = Column(

        nullable=False,
        index=True,

    description = Column(
        String(255),
        nullable=True,
    )

    is_active = Column(
        Boolean,
        default=True,
        nullable=False,
    )

        return f"<Permission {self.name}>"

from __future__ import annotations

from typing import Any

from sqlalchemy.orm import Session

from app.core.services.base_service import BaseService
from app.modules.roles.models import Role
from app.modules.roles.repositories.permission_repository import (
    permission_repository,
)
from app.modules.roles.repositories.role_repository import (
    RoleRepository,
    role_repository,
)


class RoleService(BaseService[RoleRepository]):
    def __init__(
        self,
        *,
        db: Session | None = None,
        current_user: Any | None = None,
        current_tenant: Any | None = None,
        repository: RoleRepository = role_repository,
    ) -> None:
        super().__init__(
            repository=repository,
            db=db,
            current_user=current_user,
            current_tenant=current_tenant,
        )

    def list_roles(self) -> list[Role]:
        db = self.require_db()
        repository = self.require_repository()

        roles = repository.list_with_permissions(db)

        for role in roles:
            role.users_count = len(role.users)

        return roles

    def get_by_id(
        self,
        role_id: int,
    ) -> Role | None:
        db = self.require_db()
        repository = self.require_repository()

        role = repository.get_with_permissions(
            db,
            role_id,
        )

        if role is not None:
            role.users_count = len(role.users)

        return role

    def create_role(
        self,
        *,
        name: str,
        description: str | None = None,
        permission_ids: list[int] | None = None,
        is_active: bool = True,
    ) -> Role:
        db = self.require_db()
        repository = self.require_repository()

        normalized_name = name.strip()

        if not normalized_name:
            raise ValueError(
                "Role name is required."
            )

        if repository.exists_by_name(
            db,
            normalized_name,
        ):
            raise ValueError(
                "Role name already exists."
            )

        permissions = self._get_permissions(
            permission_ids or []
        )

        with self.uow as uow:
            role = repository.create(
                db,
                data={
                    "name": normalized_name,
                    "description": (
                        description.strip()
                        if description
                        else None
                    ),
                    "is_active": is_active,
                },
            )

            role.permissions = permissions
            uow.commit()

        return self._reload_role(role.id)

    def update_role(
        self,
        role_id: int,
        *,
        name: str | None = None,
        description: str | None = None,
        is_active: bool | None = None,
        permission_ids: list[int] | None = None,
    ) -> Role:
        db = self.require_db()
        repository = self.require_repository()

        role = repository.get_with_permissions(
            db,
            role_id,
        )

        if role is None:
            raise LookupError(
                "Role was not found."
            )

        update_data: dict[str, Any] = {}

        if name is not None:
            normalized_name = name.strip()

            if not normalized_name:
                raise ValueError(
                    "Role name is required."
                )

            if repository.exists_by_name(
                db,
                normalized_name,
                exclude_role_id=role_id,
            ):
                raise ValueError(
                    "Role name already exists."
                )

            update_data["name"] = normalized_name

        if description is not None:
            update_data["description"] = (
                description.strip() or None
            )

        if is_active is not None:
            update_data["is_active"] = is_active

        permissions = None

        if permission_ids is not None:
            permissions = self._get_permissions(
                permission_ids
            )

        with self.uow as uow:
            if update_data:
                for field, value in update_data.items():
                    setattr(role, field, value)

            if permissions is not None:
                role.permissions = permissions

            uow.commit()

        return self._reload_role(role_id)

    def change_status(
        self,
        role_id: int,
        *,
        is_active: bool,
    ) -> Role:
        return self.update_role(
            role_id,
            is_active=is_active,
        )

    def update_permissions(
        self,
        role_id: int,
        *,
        permission_ids: list[int],
    ) -> Role:
        return self.update_role(
            role_id,
            permission_ids=permission_ids,
        )

    def delete_role(
        self,
        role_id: int,
    ) -> None:
        db = self.require_db()
        repository = self.require_repository()

        role = repository.get_with_permissions(
            db,
            role_id,
        )

        if role is None:
            raise LookupError(
                "Role was not found."
            )

        if role.users:
            raise ValueError(
                "Role cannot be deleted because it is assigned "
                "to one or more users."
            )

        with self.uow as uow:
            db.delete(role)
            uow.commit()

    def _get_permissions(
        self,
        permission_ids: list[int],
    ) -> list:
        db = self.require_db()

        unique_ids = list(
            dict.fromkeys(permission_ids)
        )

        permissions = permission_repository.get_by_ids(
            db,
            unique_ids,
        )

        if len(permissions) != len(unique_ids):
            found_ids = {
                permission.id
                for permission in permissions
            }

            missing_ids = sorted(
                set(unique_ids) - found_ids
            )

            raise ValueError(
                f"Permissions were not found: {missing_ids}"
            )

        return permissions

    def _reload_role(
        self,
        role_id: int,
    ) -> Role:
        db = self.require_db()
        repository = self.require_repository()

        role = repository.get_with_permissions(
            db,
            role_id,
        )

        if role is None:
            raise RuntimeError(
                "Role could not be reloaded."
            )

        role.users_count = len(role.users)

        return role


role_service = RoleService()

from sqlalchemy import func, select
from sqlalchemy.orm import Session, selectinload

from app.core.database.base_repository import BaseRepository
from app.modules.roles.models import Role
from app.shared.query import QueryOptions


class RoleRepository(BaseRepository[Role]):
    def __init__(self) -> None:
        super().__init__(
            model=Role,
            query_options=QueryOptions(
                filterable_fields={
                    "id",
                    "name",
                    "is_active",
                },
                searchable_fields={
                    "name",
                    "description",
                },
                sortable_fields={
                    "id",
                    "name",
                    "created_at",
                },
            ),
        )

    def get_with_permissions(
        self,
        db: Session,
        role_id: int,
    ) -> Role | None:
        return (
            db.execute(
                select(Role)
                .options(
                    selectinload(Role.permissions),
                    selectinload(Role.users),
                )
                .where(Role.id == role_id)
            )
            .scalar_one_or_none()
        )

    def list_with_permissions(
        self,
        db: Session,
    ) -> list[Role]:
        result = db.execute(
            select(Role)
            .options(
                selectinload(Role.permissions),
                selectinload(Role.users),
            )
            .order_by(Role.name.asc())
        )

        return list(
            result.scalars().unique().all()
        )

    def exists_by_name(
        self,
        db: Session,
        name: str,
        *,
        exclude_role_id: int | None = None,
    ) -> bool:
        statement = select(Role.id).where(
            func.lower(Role.name)
            == name.strip().lower()
        )

        if exclude_role_id is not None:
            statement = statement.where(
                Role.id != exclude_role_id
            )

        return (
            db.execute(statement).scalar_one_or_none()
            is not None
        )

    def count_users(
        self,
        db: Session,
        role_id: int,
    ) -> int:
        role = self.get_with_permissions(
            db,
            role_id,
        )

        if role is None:
            return 0

        return len(role.users)


role_repository = RoleRepository()

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.core.database.base_repository import BaseRepository
from app.modules.roles.models import Permission
from app.shared.query import QueryOptions


class PermissionRepository(BaseRepository[Permission]):
    def __init__(self) -> None:
        super().__init__(
            model=Permission,
            query_options=QueryOptions(
                filterable_fields={
                    "id",
                    "resource",
                    "action",
                    "is_active",
                },
                searchable_fields={
                    "name",
                    "description",
                    "resource",
                    "action",
                },
                sortable_fields={
                    "id",
                    "name",
                    "resource",
                    "action",
                    "created_at",
                },
            ),
        )

    def exists_by_name(
        self,
        db: Session,
        name: str,
    ) -> bool:
        return self.exists(
            db,
            name=name,
        )

    def exists_by_resource_action(
        self,
        db: Session,
        *,
        resource: str,
        action: str,
    ) -> bool:
        return self.exists(
            db,
            resource=resource,
            action=action,
        )

    def get_by_ids(
        self,
        db: Session,
        permission_ids: list[int],
    ) -> list[Permission]:
        if not permission_ids:
            return []

        unique_ids = list(
            dict.fromkeys(permission_ids)
        )

        result = db.execute(
            select(Permission).where(
                Permission.id.in_(unique_ids)
            )
        )

        return list(
            result.scalars().all()
        )


permission_repository = PermissionRepository()
